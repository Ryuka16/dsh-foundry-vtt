# 恢复资源使用次数宏

**FVTT版本V12,DND3.X系统已验证。高版本可能需要自行调整api。**

快速配置资源次数，适合与其他的物品或者宏联动。

```JavaScript
let restoreTarget = 'resources.primary.value'; // 此处输入需要恢复的资源字段
let currentResourceValue = getProperty(actor.system, restoreTarget);

let d = new Promise((resolve) => {
    let theDialog = new Dialog({
        title: '恢复资源点数',
        content: `你希望恢复多少资源？<br><input id="resourcepoints" type="number" min="0" step="1.0" value="1"></input>`,
        buttons: {
            restore: {
                label: '确认',
                icon: '<i class="fas fa-plus"></i>',
                callback: (html) => {
                    let pointsToRestore = parseInt(html.find('#resourcepoints')[0].value) || 1; // Default to 1 if no input or invalid
                    if (pointsToRestore < 1 || isNaN(pointsToRestore)) {
                        ui.notifications.warn('请输入一个有效的点数。');
                        resolve(false);
                    } else {
                        resolve(pointsToRestore);
                    }
                }
            },
            cancel: {
                label: '取消',
                icon: '<i class="fas fa-times"></i>',
                callback: () => { resolve(false); }
            }
        },
        default: 'cancel'
    }).render(true);
});

d.then((restored) => {
    if (restored !== false) {
        // Update the resource value without checking the maximum
        actor.update({[`data.${restoreTarget}`]: currentResourceValue + restored});
        ChatMessage.create({
            speaker: ChatMessage.getSpeaker({ actor: actor }),
            content: `${actor.name}资源恢复了${restored}点。`
        });
    }
}).catch((error) => console.error('发生错误：', error));
```
