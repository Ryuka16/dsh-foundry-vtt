# 消耗资源使用次数宏



**FVTT版本V12,DND3.X系统已验证。高版本可能需要自行调整api。**

快速配置资源次数，适合与其他的物品或者宏联动。

```JavaScript
let consumeTarget = 'resources.primary.value'; // 此处输入消耗的资源
const available = getProperty(actor.system, consumeTarget);

if (available <= 0) {
    ui.notifications.warn('没有资源可以使用');
    return false;
}

let d = new Promise((resolve) => {
    let theDialog = new Dialog({
        title: '消耗资源点数',
        content: `你希望消耗多少资源? ${available}可用<br><input id="resourcepoints" type="number" min="0" step="1.0" max="${available}"></input>`,
        buttons: {
            use: {
                label: '确认',
                icon: '<i class="fas fa-check"></i>',
                callback: (html) => {
                    let pointsToConsume = parseInt(html.find('#resourcepoints')[0].value);
                    if (pointsToConsume > available || isNaN(pointsToConsume)) {
                        ui.notifications.warn('输入的点数无效或超出可用资源！');
                        resolve(false);
                    } else {
                        resolve(pointsToConsume);
                    }
                }
            },
            cancel: {
                label: '取消',
                icon: '<i class="fas fa-times"></i>',
                callback: () => { resolve(false); }
            }
        },
        default: 'cancel'
    }).render(true);
});

d.then((consumed) => {
    if (consumed !== false) {
        actor.update({[`data.${consumeTarget}`]: available - consumed});
        ChatMessage.create({
            speaker: ChatMessage.getSpeaker({ actor: actor }),
            content: `${actor.name}资源减少了${consumed}点。`
        });
    }
}).catch((error) => console.error('发生错误：', error));
```
