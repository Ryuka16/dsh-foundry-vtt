将最终应用的伤害减半，可以无需AC5E。

```JavaScript
// 伤害减半宏
const workflow = args[0].workflow;
const damageItem = workflow.damageItem;

// 计算总伤害（排除 temphp 和 midi-none 类型）
const fullTotalDamage = Math.floor(
    damageItem.damageDetail.reduce((acc, d) => {
        const damageValue = d.value ?? d.damage ?? 0;
        return acc + (['temphp', 'midi-none'].includes(d.type) ? 0 : damageValue);
    }, 0)
);

// 计算减半后的伤害
const newFullDamage = Math.floor(fullTotalDamage / 2);

// 设置 HP 伤害的方式
if (!MidiQOL.configSettings().useDamageDetail) {
    // 更新伤害项的总伤害
    damageItem.appliedDamage = newFullDamage;
    damageItem.totalDamage = newFullDamage;
    
    // 如果存在旧属性也更新
    if (damageItem.hpDamage !== undefined) {
        damageItem.hpDamage = newFullDamage;
    }
}

// 创建减半伤害调整项
const halfDamage = {
    value: -fullTotalDamage + newFullDamage,
    active: { multiplier: 1 },
    type: 'midi-none',
    name: '伤害减半',
    flags: { midiQol: { isDamageAdjustment: true } }
};

// 添加到伤害详情
damageItem.damageDetail.push(halfDamage);

// 添加到详情列表
damageItem.details = damageItem.details || [];
damageItem.details.push('伤害减半');

// 更新工作流的总伤害
workflow.damageTotal = newFullDamage;

// 可选：在聊天中显示通知
if (game.settings.get("midi-qol", "AutoFastForwardDamage")) {
    const speaker = ChatMessage.getSpeaker({ token: workflow.token });
    ChatMessage.create({
        speaker: speaker,
        content: `${workflow.actor.name} 的伤害被减半：${fullTotalDamage} → ${newFullDamage}`,
        type: CONST.CHAT_MESSAGE_TYPES.EMOTE,
        whisper: [game.user.id]
    });
}

console.log(`伤害减半应用：原伤害 ${fullTotalDamage}，新伤害 ${newFullDamage}`);
```

1. 第一步：创建一个宏在世界中，命名为：`halfdamage`，将上述宏填写进去并保存
