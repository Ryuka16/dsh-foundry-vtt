# 法术运算宏

![图片展示的是FVTT系统中“法术运算”特性界面。上方显示“法术运算”图标及“特性”字样，右侧有“>”图标。下方是新角色（382）信息，法师等级为3，有“特殊”“自身”标签。中间部分显示6d6掷骰结果为\[1, 3, 4, 5, 5, 6\]，总和为24。底部列出法师基础运算情况，其中“强力对子”和“等差数列”为√，“完美数列](https://feishu.cn/file/OHYnbOCgooGjP3xAmBCcgsThnpd)

一般在私设中使用，允许配置骰子数量，并计算公式中各种效果是否符合条件，也可以计算相应数量。请根据版本和自身规则自定义修改。

![图片展示了法术运算宏在DND3.X系统中的效果说明。当掷骰出的骰子中有4以上个一样的点数，本回合内下一次施展的所需豁免的法术，目标必定豁免失败；有4个以上连续的点数，施展法术无需消耗法术位；有5个偶数或5个奇数，下一次施法的伤害骰满；骰子中有1/3/5或2/4/6时，可传送15尺；骰子总和超过24时，可再框选一片施法距离内与之前区域重合的新区域同时生效。这些效果与文档中法术运算宏的使用说明紧密相关。](https://feishu.cn/file/XCdqbWveJomumExHp0kcrIn9nog)

```JavaScript
// 配置参数 - 在这里修改骰子数量
const diceCount = 6;
const diceFaces = 6;

try {
    // 创建掷骰对象并执行 - 添加错误处理
    let roll;
    try {
        roll = new Roll(`${diceCount}d${diceFaces}`);
        await roll.evaluate({async: true});
    } catch (rollError) {
        console.warn("标准掷骰失败，使用Math.random替代:", rollError);
        // 如果Roll构造失败，使用Math.random生成结果
        let results = [];
        for (let i = 0; i < diceCount; i++) {
            results.push(Math.floor(Math.random() * diceFaces) + 1);
        }
        roll = {
            formula: `${diceCount}d${diceFaces}`,
            total: results.reduce((sum, val) => sum + val, 0),
            terms: [{
                results: results.map(result => ({ result }))
            }]
        };
    }
    
    // 获取骰子结果 - 兼容不同版本
    let diceResults = [];
    if (roll.dice && roll.dice.length > 0 && roll.dice[0].results) {
        // Foundry VTT 标准版本
        diceResults = roll.dice[0].results.map(r => r.result);
    } else if (roll.terms && roll.terms.length > 0 && roll.terms[0].results) {
        // 替代方法
        diceResults = roll.terms[0].results.map(r => r.result);
    } else if (Array.isArray(roll.terms) && roll.terms.length > 0) {
        // 再次尝试获取结果
        if (roll.terms[0] && Array.isArray(roll.terms[0])) {
            diceResults = roll.terms[0].map(r => typeof r === 'object' && r.result !== undefined ? r.result : r);
        } else {
            // 最后备选方案
            for (let i = 0; i < diceCount; i++) {
                diceResults.push(Math.floor(Math.random() * diceFaces) + 1);
            }
        }
    } else {
        // 如果以上方法都失败，使用随机数生成
        for (let i = 0; i < diceCount; i++) {
            diceResults.push(Math.floor(Math.random() * diceFaces) + 1);
        }
    }
    
    // 确保结果数组长度正确
    if (diceResults.length !== diceCount) {
        console.warn(`骰子结果数量不匹配: 期望${diceCount}, 实际${diceResults.length}, 补充随机值`);
        while (diceResults.length < diceCount) {
            diceResults.push(Math.floor(Math.random() * diceFaces) + 1);
        }
        diceResults = diceResults.slice(0, diceCount); // 确保长度不超过期望值
    }
    
    // 排序结果
    diceResults.sort((a, b) => a - b);
    let total = diceResults.reduce((sum, num) => sum + num, 0);

    // 计算所有条件 - 按顺序计算，避免冲突
    // 1. 检查是否有5个或以上连续点数 (修正：应该是5个连续，对应4个或以上连续点数)
    let hasConsecutive4 = false;
    // 去重后检查连续序列
    let uniqueSorted = [...new Set(diceResults)].sort((a, b) => a - b);
    for (let i = 0; i <= uniqueSorted.length - 4; i++) {
        let consecutive = true;
        for (let j = 1; j < 4; j++) {
            if (uniqueSorted[i + j] !== uniqueSorted[i] + j) {
                consecutive = false;
                break;
            }
        }
        if (consecutive) {
            hasConsecutive4 = true;
            break;
        }
    }

    // 2. 检查是否有4个或以上一致点数
    let countMap = {};
    diceResults.forEach(num => {
        countMap[num] = (countMap[num] || 0) + 1;
    });
    let hasSame4 = Object.values(countMap).some(count => count >= 4);

    // 3. 检查是否有5个或以上奇数或偶数
    let evenCount = diceResults.filter(num => num % 2 === 0).length;
    let oddCount = diceResults.filter(num => num % 2 === 1).length;
    let has4EvenOrOdd = evenCount >= 5 || oddCount >= 5;

    // 4. 检查是否包含1/3/5或2/4/6
    let has135 = diceResults.includes(1) && diceResults.includes(3) && diceResults.includes(5);
    let has246 = diceResults.includes(2) && diceResults.includes(4) && diceResults.includes(6);
    let hasSpecialCombo = has135 || has246;

    // 5. 检查总数是否超过24
    let hasTotalOver24 = total >= 25; // 修正为>=25，因为"超过24"

    // 新增判断（只显示√或×）
    
    // 1. 投出的1和6的总和是否≥3
    let count1 = diceResults.filter(num => num === 1).length;
    let count6 = diceResults.filter(num => num === 6).length;
    let has1And6SumGE3 = (count1 + count6) >= 3;
    
    // 2. 投出的数值是否有至少3个数字＞3
    let countGT3 = diceResults.filter(num => num > 3).length;
    let has3GT3 = countGT3 >= 3;
    
    // 3. 投出的数字是否有4个不一样的
    let uniqueCount = new Set(diceResults).size;
    let has4Different = uniqueCount >= 4;
    
    // 4. 投出的数值是否有至少3个数字≤3
    let countLE3 = diceResults.filter(num => num <= 3).length;
    let has3LE3 = countLE3 >= 3;
    
    // 5. 是否有至少3个一致的数字
    let hasSame3 = Object.values(countMap).some(count => count >= 3);
    
    // 6. 是否出现至少3个2,4或6
    let countEven = diceResults.filter(num => num === 2 || num === 4 || num === 6).length;
    let has3Even = countEven >= 3;
    
    // 7. 是否有任意3个不连续数字
    let has3NonConsecutive = false;
    if (uniqueCount >= 3) {
        let uniqueSortedLocal = [...new Set(diceResults)].sort((a, b) => a - b);
        for (let i = 0; i <= uniqueSortedLocal.length - 3; i++) {
            // 如果三个数字中至少有一对不连续
            if (uniqueSortedLocal[i+1] - uniqueSortedLocal[i] > 1 || uniqueSortedLocal[i+2] - uniqueSortedLocal[i+1] > 1) {
                has3NonConsecutive = true;
                break;
            }
        }
    }
    
    // 8. 是否结果不包括1和6
    let no1And6 = !diceResults.includes(1) && !diceResults.includes(6);
    
    // 9. 是否结果＞18
    let totalGT18 = total > 18;
    
    // 10. 是否投出的1的数量不为1
    let count1Not1 = count1 !== 1;
    
    // 11. 是否投出的结果≤18
    let totalLE18 = total <= 18;
    
    // 12. 是否有至少两个连续的两组数
    let hasTwoConsecutivePairs = false;
    let consecutiveGroups = 0;
    let usedForConsecutive = new Array(diceResults.length).fill(false);
    for (let i = 0; i < diceResults.length - 1; i++) {
        if (!usedForConsecutive[i] && diceResults[i + 1] - diceResults[i] === 1) {
            consecutiveGroups++;
            usedForConsecutive[i] = true;
            usedForConsecutive[i + 1] = true;
        }
    }
    hasTwoConsecutivePairs = consecutiveGroups >= 2;

    // 施法风格运算（显示数量）
    
    // 1. 衡鉴施法: 最大相同点数的数量
    let maxSameCount = Math.max(...Object.values(countMap), 0); // 防止空对象
    
    // 2. 粹丽施法: 成对骰子相加>6的数量
    let pairsOver6 = 0;
    let used = new Array(diceResults.length).fill(false);
    
    // 优先组合大数
    for (let i = diceResults.length - 1; i >= 0; i--) {
        if (used[i]) continue;
        
        for (let j = i - 1; j >= 0; j--) {
            if (used[j]) continue;
            
            if (diceResults[i] + diceResults[j] > 6) {
                pairsOver6++;
                used[i] = true;
                used[j] = true;
                break;
            }
        }
    }
    
    // 3. 量析施法: 2个点数相连的数量
    let consecutivePairs = 0;
    for (let i = 0; i < diceResults.length - 1; i++) {
        if (diceResults[i + 1] - diceResults[i] === 1) {
            consecutivePairs++;
        }
    }
    
    // 4. 银毫施法: 每有2个偶数记为1
    let evenPairs = Math.floor(evenCount / 2);
    
    // 5. 靡华施法: 每有2个奇数记为1
    let oddPairs = Math.floor(oddCount / 2);

    // 生成显示文本
    let output = `
<style>

.dice-formula { font-weight: bold; color: #333; margin-bottom: 5px; }
.dice-tooltip { background: #f0f0f0; padding: 5px; border-radius: 3px; margin: 5px 0; }
.dice { margin: 5px 0; }
.dice-total { font-weight: bold; color: #0066cc; font-size: 1.1em; }
h3 { color: #2c3e50; border-bottom: 1px solid #ccc; padding-bottom: 3px; margin-top: 20px; }
ul { margin: 10px 0; line-height: 1.5; }
.error { color: red; font-weight: bold; }

<div class="dice-roll">
    <div class="dice-result">
        <div class="dice-formula">${roll.formula || `${diceCount}d${diceFaces}`}</div>
        <div class="dice-tooltip">
            <div class="dice">掷骰结果: [${diceResults.join(', ')}]</div>
        </div>
        <div class="dice-total">总和: ${total}</div>
    </div>
</div>

    <li>完美数列: <span class="${hasSame4 ? 'checkmark' : 'cross'}">${hasSame4 ? '✓' : '✗'}</span></li>
    <li>幸运数差: <span class="${has4EvenOrOdd ? 'checkmark' : 'cross'}">${has4EvenOrOdd ? '✓' : '✗'}</span></li>

    <li>银毫施法: ${evenPairs}</li>

    // 发送到聊天栏 - 添加兼容性检查
    if (typeof ChatMessage !== 'undefined' && ChatMessage.create) {
```
