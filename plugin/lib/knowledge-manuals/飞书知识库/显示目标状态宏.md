# 显示目标状态宏

**FVTT版本V12,DND3.X系统已验证。高版本可能需要自行调整api。**

可以一键显示生物类型，抗性，易伤，伤害免疫，伤害调整，状态免疫等，支持自定义房规，支持自定义翻译，支持大部分客制化要求。

![图片展示了新角色（382）的属性详情。其生物类型为类人生物，伤害翻倍有光耀、雷鸣，伤害减半有强酸、钝击，伤害免疫为寒冷、毒素，伤害抗性包括活力易伤2、虚化易伤4等，状态免疫有燃烧、浸湿等。该图片与文档中介绍的显示目标状态宏功能相关，直观呈现了宏能显示的生物类型、抗性、易伤、伤害免疫、伤害调整、状态免疫等信息，帮助理解宏的显示效果。](https://feishu.cn/file/IfFsbDhhxolKtaxAYBCcJHmLnUh)

*（演示中的房规：伤害翻倍=易伤，伤害减半=抗性，伤害抗性=伤害调整；代码可以不用调整，只有有相关抗性&易伤&调整等才会显示）*

```JavaScript
// 获取用户当前选择的令牌
let token = canvas.tokens.controlled[0];
if (!token) {
    ui.notifications.warn("请先选择一个令牌。");
    return;
}

let actor = token.actor;
if (!actor) {
    ui.notifications.warn("无法找到对应的角色数据。");
    return;
}

// 直接使用system属性
let system = actor.system;

// 翻译映射表 - 保持原始大小写
const translationMap = {
    // 伤害类型
    'acid': '强酸',
    'bludgeoning': '钝击',
    'cold': '寒冷',
    'fire': '火焰',
    'force': '力场',
    'lightning': '闪电',
    'necrotic': '黯蚀',
    'piercing': '穿刺',
    'poison': '毒素',
    'psychic': '心灵',
    'radiant': '光耀',
    'slashing': '挥砍',
    'thunder': '雷鸣',
    'none': '无类型',
    'midi-none': '其它',
    'Vitality': '活力',
    'Aether': '虚化',
    'Chaos': '混乱',
    'Dragon': '龙气',
    'Basicity': '碱性',
    'Corruption': '腐败',
    'Magic': '魔力',
    
    // 状态免疫
    'blinded': '目盲',
    'charmed': '魅惑',
    'deafened': '耳聋',
    'exhaustion': '力竭',
    'frightened': '恐慌',
    'grappled': '擒抱',
    'incapacitated': '失能',
    'diseased': '疾病',
    'invisible': '隐形',
    'paralyzed': '麻痹',
    'petrified': '石化',
    'poisoned': '中毒',
    'prone': '倒地',
    'restrained': '束缚',
    'stunned': '震慑',
    'unconscious': '昏迷',
    'liuxue': '流血',
    'holy': '神圣'
};

// 翻译函数 - 保持原始大小写
function translateTerm(term) {
    if (!term || typeof term !== 'string') return term;
    
    // 直接查找翻译
    if (translationMap[term]) {
        return translationMap[term];
    }
    
    // 尝试小写查找（作为备选）
    const lowerTerm = term.toLowerCase();
    if (translationMap[lowerTerm]) {
        return translationMap[lowerTerm];
    }
    
    // 如果没有找到翻译，返回原词
    return term;
}

// 使用Array.from()处理类数组对象的函数
function getSafeArray(value) {
    // 如果是Set类型，转换为数组
    if (value instanceof Set) {
        return [...value];
    }
    if (Array.isArray(value)) {
        return value;
    }
    // 如果是类数组对象，使用Array.from()转换
    if (value && typeof value === 'object' && 'length' in value) {
        return Array.from(value);
    }
    return [];
}

// 获取生物类型
let creatureType = system.details?.type?.label || "未设置";

// 使用Array.from()处理伤害相关特性
let damageVulnerabilities = getSafeArray(system.traits?.dv?.value).map(translateTerm);
let damageResistances = getSafeArray(system.traits?.dr?.value).map(translateTerm);
let damageImmunities = getSafeArray(system.traits?.di?.value).map(translateTerm);

// 处理伤害调整对象 - 优化：反转数值以符合描述逻辑
let damageModifications = system.traits?.dm?.amount || {};
let damageModArray = [];
if (damageModifications && typeof damageModifications === 'object') {
    for (let [damageType, amount] of Object.entries(damageModifications)) {
        if (amount !== 0) {
            // 反转数值：系统中负值为减伤，正值为加伤
            // 描述中希望正值为减伤，负值为加伤
            let displayAmount = -amount;
            let effectDescription = displayAmount > 0 ? 
                `抗性 ${displayAmount}` : 
                `易伤 ${Math.abs(displayAmount)}`;
            
            damageModArray.push(`${translateTerm(damageType)}: ${effectDescription}`);
        }
    }
}

// 处理状态免疫：合并自定义值和数组值
let customConditionImmunities = (system.traits?.ci?.custom || "")
    .split(';')
    .map(s => s.trim())
    .filter(s => s !== '')
    .map(translateTerm);

let selectedConditionImmunities = getSafeArray(system.traits?.ci?.value).map(translateTerm);
let allConditionImmunities = [...customConditionImmunities, ...selectedConditionImmunities];

// 格式化显示函数
function formatTrait(traitArray) {
    if (!Array.isArray(traitArray) || traitArray.length === 0) {
        return "无";
    }
    return traitArray.join(", ");
}

// 构建聊天消息
let content = `
    <div style="
        display: grid; 
        grid-template-columns: max-content 1fr; 
        gap: 0.5rem 1rem; 
        align-items: start;
        padding: 1rem;
        background: rgba(0, 0, 0, 0.8);
        border: 2px solid #4a4a4a;
        border-radius: 8px;
        color: white;
        font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
    ">
        <strong style="
            grid-column: 1 / -1; 
            text-align: center; 
            border-bottom: 2px solid #8B4513; 
            padding-bottom: 0.5rem; 
            margin-bottom: 0.5rem;
            font-size: 1.2em;
            color: #FFD700;
        ">${actor.name} 的属性详情</strong>
        
        <strong style="color: #87CEEB;">生物类型:</strong>
        <span>${creatureType}</span>
        
        <strong style="color: #FF6B6B;">伤害翻倍:</strong>
        <span>${formatTrait(damageVulnerabilities)}</span>
        
        <strong style="color: #4ECDC4;">伤害减半:</strong>
        <span>${formatTrait(damageResistances)}</span>
        
        <strong style="color: #FFA500;">伤害免疫:</strong>
        <span>${formatTrait(damageImmunities)}</span>
        
        <strong style="color: #9B59B6;">伤害抗性:</strong>
        <span>${damageModArray.length > 0 ? damageModArray.join(", ") : "无"}</span>
        
        <strong style="color: #2ECC71;">状态免疫:</strong>
        <span>${formatTrait(allConditionImmunities)}</span>
    </div>
`;

// 发送聊天消息
let chatData = {
    user: game.user.id,
    content: content,
    speaker: ChatMessage.getSpeaker({ token: actor })
};

ChatMessage.create(chatData);
```
