# 自定义豁免属性宏

**FVTT版本V12,DND3.X系统已验证。高版本可能需要自行调整api。**

支持自行选择目标所需进行的豁免（无后续检测，可以自行配置），自动读取当前选中演员的施法DC（可以固定DC），支持批量目标的豁免。

```JavaScript
// 目标演员豁免检定宏 
const abilities = {
  "str": "力量豁免",
  "dex": "敏捷豁免", 
  "con": "体质豁免",
  "int": "智力豁免",
  "wis": "感知豁免",
  "cha": "魅力豁免"
};

// 获取目标令牌
function getTargetTokens() {
  let targets = [];
  
  // 方法1: 使用game.user.targets获取目标
  if (game.user.targets && game.user.targets.size > 0) {
    targets = Array.from(game.user.targets);
    console.log(`使用game.user.targets获取到 ${targets.length} 个目标`);
  }
  
  // 方法2: 使用Monk's TokenBar获取目标 (备用)
  if (targets.length === 0 && game.modules.get("monks-tokenbar")?.active && game.MonksTokenBar?.getTargets) {
    targets = game.MonksTokenBar.getTargets();
    if (targets.length > 0) {
      console.log(`使用Monk's TokenBar获取到 ${targets.length} 个目标`);
    }
  }
  
  // 方法3: 如果没有目标，使用选中的令牌作为备用
  if (targets.length === 0 && canvas.tokens.controlled.length > 0) {
    ui.notifications.warn("未找到目标，将使用当前选中的令牌");
    targets = canvas.tokens.controlled;
  }
  
  return targets;
}

// 获取施法者DC
function getCasterDC(casterActor) {
  return casterActor.system.attributes?.spelldc?.value || 
         casterActor.system.abilities?.cha?.dc || 
         10;
}

// 执行豁免检定的函数
async function executeSavingThrow(token, ability, dc) {
  const actor = token.actor;
  if (!actor) return false;
  
  const abilityName = abilities[ability] || ability;
  
  try {
    // 方法1: 使用midi-qol
    if (game.modules.get("midi-qol")?.active && game.midiQOL?.abilityRoll) {
      const options = {
        rollType: "save",
        ability: ability,
        dc: dc,
        flavor: `${actor.name} 进行${abilityName}豁免 (DC ${dc})`
      };
      
      await game.midiQOL.abilityRoll(actor, options);
      return true;
    }
    
    // 方法2: 使用Monk's Token Bar
    if (game.modules.get("monks-tokenbar")?.active && game.MonksTokenBar?.abilitySave) {
      await game.MonksTokenBar.abilitySave(actor, ability, {
        dc: dc,
        chatMessage: true
      });
      return true;
    }
    
    // 方法3: 使用核心系统
    if (actor.rollAbilitySave) {
      await actor.rollAbilitySave(ability, {
        flavor: `豁免检定: ${abilityName} (DC ${dc})`,
        chatMessage: true
      });
      return true;
    }
    
    return false;
  } catch (error) {
    console.error(`为 ${actor.name} 执行豁免检定时出错:`, error);
    return false;
  }
}

// 主执行函数
async function main() {
  // 获取施法者（当前选中的演员）
  const casterTokens = canvas.tokens.controlled;
  if (casterTokens.length === 0) {
    ui.notifications.warn("请先选择一个施法者令牌！");
    return;
  }
  
  const casterActor = casterTokens[0].actor;
  const spellDC = getCasterDC(casterActor);
  
  // 获取目标令牌
  const targetTokens = getTargetTokens();
  if (targetTokens.length === 0) {
    ui.notifications.error("未找到任何目标令牌！请先设置目标。");
    return;
  }
  
  // 创建对话框选择豁免属性
  new Dialog({
    title: "批量豁免检定",
    content: `
      <div style="margin-bottom: 10px; padding: 10px; background: rgba(0,0,0,0.05); border-radius: 5px;">
        <div style="display: flex; justify-content: space-between; margin-bottom: 8px;">
          <span><strong>施法者:</strong> ${casterActor.name}</span>
          <span><strong>目标数:</strong> ${targetTokens.length}</span>
        </div>
        <div style="display: flex; justify-content: space-between;">
          <span><strong>施法DC:</strong> ${spellDC}</span>
          <span><i class="fas fa-bullseye"></i></span>
        </div>
      </div>
      <form>
        <div class="form-group">
          <label style="font-weight: bold;">选择豁免属性:</label>
          <select id="abilitySelect" name="abilitySelect" style="margin-top: 5px;">
            <option value="str">力量豁免</option>
            <option value="dex">敏捷豁免</option>
            <option value="con">体质豁免</option>
            <option value="int">智力豁免</option>
            <option value="wis">感知豁免</option>
            <option value="cha">魅力豁免</option>
          </select>
        </div>
      </form>
      <div style="margin-top: 10px; font-size: 0.8em; color: #666;">
        <i class="fas fa-info-circle"></i> 将对 ${targetTokens.length} 个目标执行豁免检定
      </div>
    `,
    buttons: {
      execute: {
        icon: '<i class="fas fa-bolt"></i>',
        label: `执行豁免 (${targetTokens.length}目标)`,
        callback: async (html) => {
          const ability = html.find('[name="abilitySelect"]').val();
          const abilityName = abilities[ability];
          
          // 显示开始提示
          ui.notifications.info(`开始对 ${targetTokens.length} 个目标执行${abilityName}...`);
          
          let successCount = 0;
          let failCount = 0;
          
          // 为每个目标执行豁免检定
          for (const token of targetTokens) {
            const success = await executeSavingThrow(token, ability, spellDC);
            if (success) {
              successCount++;
            } else {
              failCount++;
            }
            
            // 短暂延迟以避免请求过载
            await new Promise(resolve => setTimeout(resolve, 100));
          }
          
          // 显示总结信息
          if (failCount === 0) {
            ui.notifications.success(`已成功为 ${successCount} 个目标执行${abilityName}豁免检定！`);
          } else {
            ui.notifications.warn(`豁免检定完成：${successCount} 成功，${failCount} 失败`);
          }
        }
      },
      cancel: {
        icon: '<i class="fas fa-times"></i>',
        label: "取消"
      }
    },
    default: "execute",
    close: () => {}
  }).render(true);
}

// 执行主函数
main().catch(console.error);
```
