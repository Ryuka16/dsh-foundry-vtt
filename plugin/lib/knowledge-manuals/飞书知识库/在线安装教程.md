# 使用清单地址

有时你需要一些稍旧版本/特定版本的mod，或是你要安装的mod虽然仓库位于GitHub，但并没有上架官方的mod商店，你会需要使用清单地址来安装mod，同样的，以这种形式安装的mod系统也会帮助你安装对应的依赖项mod（如果有）。

首先，你需要找到对应的清单地址，我们有以下几种方法：

## 官网

你需要前往官网的mod网址，比如https://foundryvtt.com/packages/midi-qol，然后下拉，到最下面的Avalible Versions，你会看到一系列版本号，以及他们支持的核心版本，Mainifest URL与Read Notes（更新日志）。找到你要安装的版本

# 通过脚本批量安装

你可以在主界面打开控制台，输入以下脚本来批量安装mod，但需要你提供对应的清单地址，同样地，系统也会自动弹出对话框询问你是否安装依赖项：

```JavaScript
(async () => {
  const manifests = [
    "https://example.com/module-a/module.json",   //替换为你要安装的mod的清单地址
    "https://example.com/module-b/module.json",
    "https://example.com/module-c/module.json"
  ];

  const results = [];

  for (const manifest of manifests) {
    try {
      console.log(`正在安装：${manifest}`);

      const pkg = await game.installPackage({
        manifest,
        notify: true
      });

      results.push({
        id: pkg.id,
        title: pkg.title,
        status: "installed"
      });
    } catch (error) {
      console.error(`安装失败：${manifest}`, error);
      results.push({
        manifest,
        status: "failed",
        error: error.message
      });
    }
  }

  console.table(results);
})();
```

如果你只有mod的id，也可以使用脚本来批量安装，前提是他们已经上架官方的mod列表

```JavaScript
(async () => {
  const ids = [
    "ID1",               //替换为你要安装的mod ID
    "ID2",
    "ID3"
  ];

  const catalog = await game.getPackages({type: "module"});
  const results = [];

  for (const id of ids) {
    if (game.modules.has(id)) {
      console.log(`已安装，跳过：${id}`);
      results.push({id, status: "skipped"});
      continue;
    }

    const info = catalog.get(id);

    if (!info) {
      console.error(`官方目录中找不到：${id}`);
      results.push({id, status: "not-found"});
      continue;
    }

    try {
      const pkg = await game.installPackage({
        id,
        manifest: info.manifest,
        notify: true
      });

      results.push({
        id: pkg.id,
        title: pkg.title,
        status: "installed"
      });
    } catch (error) {
      console.error(`安装失败：${id}`, error);
      results.push({id, status: "failed", error: error.message});
    }
  }

  console.table(results);
})();
```

每次都需要更改脚本中的`ids`或是`manifests`可能会稍显庞杂，你也可以在本地维护一个特殊的json，然后读取json里的`ids`/`manifests`来进行批量安装，此时脚本如下

```TypeScript
(async () => {
  // 打开本地 JSON 文件
  const file = await new Promise(resolve => {
    const input = document.createElement("input");
    input.type = "file";
    input.accept = ".json,application/json";
    input.onchange = () => resolve(input.files?.[0]);
    input.click();
  });

  if (!file) {
    console.warn("没有选择文件");
    return;
  }

  let json;
  try {
    json = JSON.parse(await file.text());
  } catch (error) {
    console.error("JSON 格式错误：", error);
    ui.notifications.error(`JSON 格式错误：${error.message}`);
    return;
  }

  // 同时支持 [...]
  // 以及 { "packages": [...] }、{ "modules": [...] }
  const entries = Array.isArray(json)
    ? json
    : json.packages ?? json.modules;

  if (!Array.isArray(entries)) {
    throw new Error(
      'JSON 顶层必须是数组，或包含 "packages"/"modules" 数组'
    );
  }

  // 标准化条目
  const packages = entries.map(entry => {
    if (typeof entry === "string") {
      return /^https?:\/\//i.test(entry)
        ? {type: "module", manifest: entry}
        : {type: "module", id: entry};
    }

    return {
      type: entry.type ?? "module",
      id: entry.id,
      manifest: entry.manifest
    };
  });

  const collections = {
    module: game.modules,
    system: game.systems,
    world: game.worlds
  };

  // 只在存在纯 ID 条目时读取对应官方目录
  const catalogs = {};

  for (const type of new Set(
    packages.filter(p => p.id && !p.manifest).map(p => p.type)
  )) {
    console.log(`正在读取 ${type} 官方目录……`);
    catalogs[type] = await game.getPackages({type});
  }

  const results = [];

  for (const entry of packages) {
    const {type, id} = entry;
    let {manifest} = entry;

    if (!["module", "system", "world"].includes(type)) {
      console.error(`不支持的包类型：${type}`);
      results.push({type, id, manifest, status: "invalid-type"});
      continue;
    }

    // 已安装则跳过
    if (id && collections[type]?.has(id)) {
      console.log(`已安装，跳过：${type}/${id}`);
      results.push({type, id, status: "skipped"});
      continue;
    }

    // 根据 ID 从官方目录取得 manifest
    if (!manifest && id) {
      const catalogEntry = catalogs[type]?.get(id);

      if (!catalogEntry) {
        console.error(`官方目录中找不到：${type}/${id}`);
        results.push({type, id, status: "not-found"});
        continue;
      }

      manifest = catalogEntry.manifest;

      if (!manifest) {
        console.error(`包没有可用 manifest：${type}/${id}`);
        results.push({type, id, status: "no-manifest"});
        continue;
      }
    }

    if (!manifest) {
      console.error("条目缺少 id 或 manifest：", entry);
      results.push({...entry, status: "invalid-entry"});
      continue;
    }

    try {
      console.log(`正在安装：${type}/${id ?? manifest}`);

      const installed = await game.installPackage({
        type,
        id,
        manifest,
        notify: true,
        force: false
      });

      results.push({
        type,
        id: installed.id,
        title: installed.title,
        status: "installed"
      });
    } catch (error) {
      console.error(`安装失败：${id ?? manifest}`, error);
      results.push({
        type,
        id,
        manifest,
        status: "failed",
        error: error.message
      });
    }
  }

  console.table(results);

  // 保留结果，之后可在控制台查看
  window.fvttBatchInstallResults = results;

  const installed = results.filter(r => r.status === "installed").length;
  const skipped = results.filter(r => r.status === "skipped").length;
  const failed = results.length - installed - skipped;

  ui.notifications.info(
    `批量处理完成：安装 ${installed}，跳过 ${skipped}，其他 ${failed}`
  );

  return results;
})();
```
