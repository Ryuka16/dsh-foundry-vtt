## 一 · 一次拿全的脚本（可复用）

```js
// 按 07 表的三条规则：反向白名单 + flags 全留 + 三层都走
const 丢键 = ["_stats", "ownership", "folder", "sort", "_source", "_key"];
const 是描述 = (p) => /(^|\.)description(\.value)?$/.test(p) || /\.description$/.test(p) || /(^|\.)biography(\.value)?$/.test(p);
const 去HTML = (s) => String(s == null ? "" : s)
  .replace(/<br\s*\/?>/gi, String.fromCharCode(10)).replace(/<\/p>/gi, String.fromCharCode(10))
  .replace(/<[^>]+>/g, "").replace(/&nbsp;/g, " ").replace(/&amp;/g, "&")
  .replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/[ \t]+/g, " ").trim();
const 清洗 = (x, path) => {
  if (x === null || x === undefined) return x;
  if (Array.isArray(x)) return x.map((v) => 清洗(v, path + "[]"));
  if (typeof x === "object") {
    const o = {};
    for (const k of Object.keys(x)) { if (丢键.indexOf(k) >= 0) continue; o[k] = 清洗(x[k], path ? path + "." + k : k); }
    return o;
  }
  if (typeof x === "string" && 是描述(path)) return 去HTML(x);
  return x;
};
const p = game.packs.get("your-dm-toolkit.your-actors");
const d = await p.getDocument("aDsNZgRvKdZykbk7");
const s = JSON.stringify(清洗(d.toObject(), ""));
return JSON.stringify({ 原始字节: JSON.stringify(d.toObject()).length, 清洗后字节: s.length, 数据: s });
```

## 三 · 7 件内嵌物品逐件解剖

### ① 长剑 Longsword `hydarnesSword001`（weapon）
- 活动：`attack`，伤害 `includeBase:true` + `parts:[{number:3, denomination:6, types:["necrotic"]}]` ⇒ **2d8 基础（长剑）+ 3d6 黯蚀**
- **`act.flags.autoanimations`**：
  - `primary.video.customPath` = `blfx.weapon.melee.longsword1.attack1.color1`
  - `primary.sound.file` = `psfx.weapon-attacks.sword.v1`
  - `secondary.video.customPath` = `jb2a.impact.dark_red.0`（命中特效）
  - `menu:"melee"` / `label:"血疟长剑"` / `version:5`

### ② 渴血斩 Bloodthirst Slash `hydarnesSword002`（weapon）★ 写法 2
- 活动：`attack`，3d6 necrotic（同上）
- **三处 flags 同时写**（少一处就静默不触发）：
```js
flags["midi-qol"].onUseMacroName = "[postActiveEffects]ItemMacro"
flags.itemacro.macro = { name:"渴血斩·汲血", type:"script", scope:"global", command:"…" }
flags.dae.macro      = { name:"渴血斩·汲血", type:"script", scope:"global", command:"…" }  // 与上一处逐字相同
```
- **宏体逻辑**（可照抄的「造伤 → 转成资源」模板）：
```js
const wf = (typeof workflow !== "undefined" && workflow) ? workflow : MidiQOL.currentWorkflow;
if (!wf) return;
if (!wf.hitTargets || wf.hitTargets.size === 0) return;      // 没命中直接结束
let necro = 0;
const detail = wf.damageDetail || [];
for (const d of detail) {
  const t = d.type || d.damageType || "";
  if (t === "necrotic") necro += Number(d.value || 0);        // ★ 只统计指定伤害类型
}
if (necro <= 0) return;
const me = wf.actor;
if (!me) return;
const hp = me.system.attributes.hp;
const cap = hp.max + (hp.tempmax || 0);                       // ★ 上限含临时最大值
const next = Math.min(cap, hp.value + necro);                 // ★ 不过量回复
if (next > hp.value) {
  await me.update({ "system.attributes.hp.value": next });
  await ChatMessage.create({
    content: "<p><strong>渴血斩</strong>：长生军汲取了 " + necro + " 点生命值。</p>",
    speaker: ChatMessage.getSpeaker({ actor: me })
  });
}
```
> **三个可直接沿用的要点**：① 用 `wf.damageDetail` 按**伤害类型**过滤，而不是读描述里的骰子 ② 回血上限含 `tempmax` ③ 真改了值才发播报（避免空播报）。

### ③ 死疫汲能 Plague Drain（feat）★ 写法 3
- `uses: { max:"1", recovery:[{ formula:"6", period:"recharge", type:"recoverAll" }], spent:0 }`
  ⇒ **充能 6**：每回合开始掷 d6，出 6 就恢复。这正是描述里写的机制。
- 描述完整交代七件事：动作 / 30 尺 / 可见且非构装非不死 / **DC 16 体质豁免** / 6d6 黯蚀 + 等量生命上限削减 / 时限内再受黯蚀伤害同样削上限 / 上限归 0 则转化为「苏丹行尸」
- flags：`dae.macro` + `itemacro.macro` + `midi-qol.onUseMacroName`（同写法 2）+ `act.flags.autoanimations`

### ④ 多重攻击 Multiattack `hydarnesMulti001`（feat）
- `utility` 活动，纯描述（「三次长剑攻击，其中任意两次可替换为渴血斩；若死疫汲能已充能，可额外使用一次」）
- **没有硬做自动化** —— 这类「组合动作」用描述交代，是**合理选择**，不是缺失

### ⑤ 不死本质 Unliving Nature `hydarnesUnlive01`（feat）
- `activation.type:"special"` + `utility`，纯描述（不需空气/食物/饮水/睡眠）
- 无字段可实现 ⇒ 空转是正常的

### ⑥ 魔鬼造物 Devil's Creation `hydarnesDevil001`（feat）
- 同上，`utility` 纯描述（免疫驱散不死效应 / 魔法黑暗不阻黑暗视觉）

### ⑦ 方阵行军 Phalanx March `hydarnesPhalanx1`（feat）★ 写法 4（最复杂的一件）
- 物品层：`flags.walledtemplates`（11 个子键：`wallsBlock` / `wallRestriction` / `snapCenter` / `snapCorner` / `snapSideMidpoint` / `hideBorder` / `hideHighlighting` / `showOnHover` / `noAutotarget` / `addTokenSize` / `attachToken`）
- `utility` 活动，纯描述（10 尺内有另一个不死盟友 ⇒ 双方 AC +3、先攻 +3，不可叠加）

**真正的实现全在那条顶层效果上**：

```js
// effects[0]
_id: "hydarnesAura0001"
type: "auraeffects.aura"          // ★ Aura Effects 模块的光环类型（不是 ActiveAuras！）
transfer: true                     // 常驻
system: {
  applyToSelf: true,               // 自己也吃
  distanceFormula: "10",           // 半径 10 尺
  disposition: 0,                  // 中立（不限阵营）
  collisionTypes: ["move"],
  color: "#8b1a1a", opacity: 0.3, showRadius: true, canStack: false,
  evaluatePreApply: true,          // ★ 施法前先评估脚本
  script: "…30 行空间判定…"        // ★★ 见下
},
changes: [
  { key: "system.attributes.ac.bonus",   mode: 2, value: "+3", priority: 20 },
  { key: "system.attributes.init.bonus", mode: 2, value: "+3", priority: 20 }
],
flags: { chris-premades: { embeddedMacros: [ …四条… ] } }
```

> ⚠️⚠️ **两个救命字段（范本原件里是隐藏的，2026-10-08 实测补上）**
> `system` 里还有 **`bestFormula`** 与 **`overrideName`** 两个键（范本原值是字符串 `"false"`）。
> **它们必须是空字符串 `""`，绝不能传布尔 `false`。**
> 原因在 Aura Effects 的 `modules/auraeffects/scripts/queries.mjs` → `applyAuraEffects`：
> ```js
> name: effect.system.overrideName?.trim() || effect.name,
> const bestValue = new Roll(effect.system.bestFormula.trim() || "0", effect.parent?.getRollData?.()).evaluateSync().total;
> ```
> dnd5e 的 StringField 会把布尔 `false` 存成**字符串 `"false"`** ⇒ `new Roll("false")` **解析失败抛错** ⇒
> **整个光环挂载流程崩掉，而且一个报错都看不到**（socket query 侧的错误不冒到调用方），现象是「光环摆在那儿但谁身上都不生效」。
> ⇒ **正确的 system 完整版**：
> ```js
> system: { applyToSelf: false, bestFormula: "", collisionTypes: ["move"], color: "#4a5a2a",
>           combatOnly: false, disableOnHidden: false, disposition: -1, distance: 5, distanceFormula: "5",
>           evaluatePreApply: true, opacity: 0.3, overrideName: "", script: "", showRadius: true, canStack: false }
> ```
> 另两条：① `addRemoveEffect` **只在 `createActiveEffect` / `deleteActiveEffect` hook 上跑** ⇒ 改字段**不会**重跑，要重跑就**删掉重建**；② `createEmbeddedDocuments` 时 `transfer: true` 会被吃成 `false`，用 `update({transfer:true})` 可改回。
> 读值判据用 **`effect.toObject().changes`** —— `e.changes` getter 会被 Aura Effects 的 mixin 藏起来读成 0，那是**正常的、不是丢数据**。
> 详见 `01_跑团工具\FVTT技术资料\dnd5e_classpack-三件光环做成-最终版-2026-10-08.md`。

**`system.script`（空间判定，可照抄的写法）**：
```js
const cand = actor;                                    // 候选（被施加光环者）
const src  = sourceToken && sourceToken.document && sourceToken.document.actor;
const S = sourceToken && sourceToken.document, T = token && token.document;
const isUndead = (a) => !!(a && a.system && a.system.details && a.system.details.type && a.system.details.type.value === "undead");
if (!isUndead(cand) || !isUndead(src) || !S || !T) return false;
const g = S.parent && S.parent.grid;
if (!g || !S.parent.tokens) return false;
const near = (u, v) => {
  const a = u.getOccupiedGridSpaceOffsets(), b = v.getOccupiedGridSpaceOffsets();
  for (const p of a) for (const q of b) {
    const pa = g.getCenterPoint(p), pb = g.getCenterPoint(q);
    if (Math.hypot(pa.x - pb.x, pa.y - pb.y) / g.size * g.distance <= 10.0001) return true;   // ★ 像素→尺换算
  }
  return false;
};
return S.parent.tokens.some(x => x !== T && isUndead(x.actor) && near(S, x));
```

**四条 CPR 内嵌宏**（`flags.chris-premades.embeddedMacros`）—— 这是**动态增删 changes** 的关键：

| 名称 | type / pass | 干什么 |
|---|---|---|
| `Phalanx self move` | `movement` / **`moved`** | 自己移动后重算 |
| `Phalanx ally near` | `movement` / **`movedNear`** | 别人走近后重算 |
| `Phalanx delivered buff` | `effect` / **`created`** | 光环效果被创建时写入两条 changes |
| `Phalanx aura left` | `effect` / **`deleted`** | 光环离开时清空 changes |

四条宏体**逻辑相同**：查 10 尺内有无「活着的不死盟友」→ **有则写两条 changes，无则 `changes: []`**，并记 `flags.auraeffects.lastSync`：

```js
const tk = trigger.token;
const me = tk && tk.actor;
if (!me) return;
const ef = me.effects.get("hydarnesAura0001");
if (!ef) return;
const al = tokenUtils.findNearby(tk, 10, "all").some(function (t) {
  const a = t.actor;
  return t.document.uuid !== tk.document.uuid && a && a.system && a.system.details && a.system.details.type
      && a.system.details.type.value === "undead" && !a.statuses.has("dead");
});
const cu = (ef.changes || []).length > 0;
if (al === cu) return;                                   // ★ 状态没变就不写（防抖动）
await genericUtils.update(ef, { changes: al ? [
  { key: "system.attributes.ac.bonus",   mode: 2, value: "+3", priority: 20 },
  { key: "system.attributes.init.bonus", mode: 2, value: "+3", priority: 20 }
] : [] });
await me.update({ "flags.auraeffects.lastSync": Date.now() });
```
> **两个可沿用的要点**：① 光环的**条件性加值**不能写死在 `changes` 里，要走「宏动态增删 changes」② `if (al === cu) return` 这行是防抖，**别省**。

## 四 · 三层载体在这件上的实际分布（对照 07 表）

| 层 | 本件出现了什么 |
|---|---|
| **顶层 `flags`** | 只有 `flags.scene-packer`（来源指纹，无功能） |
| **效果层 `eff.flags`** | `auraeffects.originalType`、**`chris-premades.embeddedMacros` ×4** |
| **活动层 `act.flags`** | **`autoanimations` ×3**（长剑 / 渴血斩 / 死疫汲能） |

⇒ **三层各有各的活**。只扫顶层会以为这件「除了 scene-packer 什么都没有」，只扫效果层会漏掉全部动画配置。

## 五 · 与 classpack 的对照（观察，不是结论）

- classpack 7 包（4,654 件）的活动层 flags **只有 `region-attacher` ×3**（困惑术 / 火球术 / 艾伐黑触手），**没有任何一件把 AA 动画挂到活动层**。
- classpack 的 AA 配置（若有）挂在别处或未配。**「要不要补」属产品决策，不是数据错误** —— 不要把它写进错误清单。

## 六 · 用这件当范本时的自检

1. 我写的宏，**三处 flags 都写了吗**？（`dae.macro` + `itemacro.macro` + `midi-qol.onUseMacroName`）
2. 我的 `onUseMacroName` pass **写对了吗**？（造伤后动作 = `postActiveEffects`）
3. 有条件性加值时，我是不是**写死在 `changes` 里**了？（应该走 CPR 宏动态增删）
4. AA 动画我挂**活动层**了吗？（挂物品层只有部分场景生效）
5. 充能类特性我用 `recovery:[{formula:"6",period:"recharge"}]` 了吗？（不是写成 `uses.max`）
